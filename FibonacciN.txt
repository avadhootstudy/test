
#include <iostream>
using namespace std;

// Recursive function
int fibonacci(int n) {
    if (n <= 1)
        return n;

    return fibonacci(n - 1) + fibonacci(n - 2);
}

// Non-recursive function
void fibonacciIterative(int n) {
    int a = 0, b = 1, c;

    for (int i = 0; i < n; i++) {
        cout << a << " ";
        c = a + b;
        a = b;
        b = c;
    }
}

int main() {
    int n;
    cout << "Enter number of terms: ";
    cin >> n;

    cout << "Fibonacci using Recursion: ";
    for (int i = 0; i < n; i++) {
        cout << fibonacci(i) << " ";
    }

    cout << "\nFibonacci without Recursion: ";
    fibonacciIterative(n);

    return 0;
}
