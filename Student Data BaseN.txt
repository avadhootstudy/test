
#include <iostream>
#include <algorithm>
using namespace std;

struct Student {
    int roll;
    string name;
    float sgpa;
};

int main() {
    Student s[15], temp;
    int n = 15, choice;
    
    for(int i = 0; i < n; i++) {
        cout << "\nEnter Roll No, Name and SGPA: ";
        cin >> s[i].roll >> s[i].name >> s[i].sgpa;
    }

    do {
        cout << "\n\n1.Roll Call (Bubble Sort)";
        cout << "\n2.Alphabetical (Insertion Sort)";
        cout << "\n3.Top 10 Toppers (Quick Sort)";
        cout << "\n4.Search by SGPA";
        cout << "\n5.Search by Name";
        cout << "\n6.Exit";
        cout << "\nEnter choice: ";
        cin >> choice;

        if(choice == 1) {
            for(int i = 0; i < n-1; i++)
                for(int j = 0; j < n-i-1; j++)
                    if(s[j].roll > s[j+1].roll)
                        swap(s[j], s[j+1]);
        }

        else if(choice == 2) {
            for(int i = 1; i < n; i++) {
                temp = s[i];
                int j = i-1;
                while(j >= 0 && s[j].name > temp.name) {
                    s[j+1] = s[j];
                    j--;
                }
                s[j+1] = temp;
            }
        }

        else if(choice == 3) {
            sort(s, s+n, [](Student a, Student b) {
                return a.sgpa > b.sgpa;
            });
        }

        else if(choice == 4) {
            float key;
            bool found = false;
            cout << "Enter SGPA: ";
            cin >> key;
            for(int i = 0; i < n; i++)
                if(s[i].sgpa == key) {
                    cout << s[i].roll << " " << s[i].name << " " << s[i].sgpa << endl;
                    found = true;
                }
            if(!found) cout << "Student not found";
            continue;
        }

        else if(choice == 5) {
            string key;
            bool found = false;
            cout << "Enter Name: ";
            cin >> key;

            // Sort alphabetically
            for(int i = 1; i < n; i++) {
                temp = s[i];
                int j = i-1;
                while(j >= 0 && s[j].name > temp.name) {
                    s[j+1] = s[j];
                    j--;
                }
                s[j+1] = temp;
            }

            // Binary Search
            int low = 0, high = n-1;
            while(low <= high) {
                int mid = (low+high)/2;
                if(s[mid].name == key) {
                    int i = mid;
                    while(i >= 0 && s[i].name == key) i--;
                    for(i++; i < n && s[i].name == key; i++) {
                        cout << s[i].roll << " " << s[i].name << " " << s[i].sgpa << endl;
                        found = true;
                    }
                    break;
                }
                else if(s[mid].name < key) low = mid+1;
                else high = mid-1;
            }
            if(!found) cout << "Student not found";
            continue;
        }
        else if(choice == 6) break;
        else {
            cout << "Invalid choice";
            continue;
        }

        // Display records
        int limit = (choice == 3) ? 10 : n;
        for(int i = 0; i < limit; i++)
            cout << s[i].roll << " " << s[i].name << " " << s[i].sgpa << endl;

    } while(choice != 6);

    return 0;
}
