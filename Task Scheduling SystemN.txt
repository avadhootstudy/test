
#include <iostream>
using namespace std;

int q[10], front = -1, rear = -1;

void addTask(int task) {
    if (rear == 9)
        cout << "Queue is Full\n";
    else {
        if (front == -1) front = 0;
        q[++rear] = task;
        cout << "Task Added\n";
    }
}

void executeTask() {
    if (front == -1 || front > rear)
        cout << "No Tasks Available\n";
    else
        cout << "Executing Task: " << q[front++] << endl;
}

void display() {
    if (front == -1 || front > rear)
        cout << "Queue is Empty\n";
    else {
        for (int i = front; i <= rear; i++)
            cout << q[i] << " ";
        cout << endl;
    }
}

int main() {
    int ch, task;
    do {
        cout << "\n1.Add Task\n2.Execute Task\n3.Display\n4.Exit\n";
        cin >> ch;

        switch (ch) {
            case 1:
                cout << "Enter Task ID: ";
                cin >> task;
                addTask(task);
                break;
            case 2: executeTask(); break;
            case 3: display(); break;
        }
    } while (ch != 4);

    return 0;
}
